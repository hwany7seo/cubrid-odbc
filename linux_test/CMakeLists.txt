message(STATUS "=========== Linux ODBC Test ===========")

CMAKE_MINIMUM_REQUIRED(VERSION 3.21)

PROJECT(cubrid-odbc-test C)

set(TARGET odbc_test)
set(TESTCASE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/testcases)

# Every testcases/<group>/sql_*.c file is one testcase, and its entry function
# has the same name as the file. Adding a new file is enough to run it.
file(GLOB_RECURSE CUBRID_ODBC_TESTCASES
  RELATIVE ${TESTCASE_DIR}
  CONFIGURE_DEPENDS
  ${TESTCASE_DIR}/*/sql_*.c
  )
list(SORT CUBRID_ODBC_TESTCASES)

if(NOT CUBRID_ODBC_TESTCASES)
  message(FATAL_ERROR "No testcase found in ${TESTCASE_DIR}/<group>/sql_*.c")
endif()

set(CUBRID_ODBC_TEST_SOURCES
  test_util.c
  odbc_debug.c
  )

# testcases.lst tells odbc_test which testcases were built into the test library
set(TESTCASE_LIST_FILE ${CMAKE_CURRENT_BINARY_DIR}/testcases.lst)
set(TESTCASE_LIST_CONTENTS "# generated by cmake: <group> <testcase>\n")

foreach(testcase ${CUBRID_ODBC_TESTCASES})
  list(APPEND CUBRID_ODBC_TEST_SOURCES ${TESTCASE_DIR}/${testcase})

  get_filename_component(testcase_group ${testcase} DIRECTORY)
  get_filename_component(testcase_name ${testcase} NAME_WE)
  string(APPEND TESTCASE_LIST_CONTENTS "${testcase_group} ${testcase_name}\n")
endforeach()

file(WRITE ${TESTCASE_LIST_FILE} "${TESTCASE_LIST_CONTENTS}")
message(STATUS "TESTCASES = ${TESTCASE_LIST_FILE}")

# unixODBC driver manager. The one built together with the driver
# (<source>/build/external) is used unless UNIXODBC_DIR is given.
set(UNIXODBC_DIR "" CACHE PATH "unixODBC install directory (holds include/sql.h and lib/libodbc.so)")

if(NOT UNIXODBC_DIR AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/../build/external/include/sql.h)
  set(UNIXODBC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../build/external)
endif()

find_path(UNIXODBC_INCLUDE_DIR sql.h HINTS ${UNIXODBC_DIR}/include)
find_library(UNIXODBC_LIBRARY NAMES odbc HINTS ${UNIXODBC_DIR}/lib ${UNIXODBC_DIR}/lib64)

if(NOT UNIXODBC_INCLUDE_DIR OR NOT UNIXODBC_LIBRARY)
  message(FATAL_ERROR "unixODBC not found. Build the driver first (build.sh) or set -DUNIXODBC_DIR=<unixODBC install directory>")
endif()

message(STATUS "UNIXODBC_INCLUDE_DIR = ${UNIXODBC_INCLUDE_DIR}")
message(STATUS "UNIXODBC_LIBRARY = ${UNIXODBC_LIBRARY}")

set (CMAKE_EXE_LINK_FLAGS "-m64 -Wall")
set (CMAKE_include_directories_BEFORE ON)

ADD_COMPILE_OPTIONS(-m64 -Wall)

INCLUDE_DIRECTORIES(${CMAKE_CURRENT_SOURCE_DIR} include ${UNIXODBC_INCLUDE_DIR})

ADD_LIBRARY(${CMAKE_PROJECT_NAME} SHARED ${CUBRID_ODBC_TEST_SOURCES})

target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE -DGCC -DLINUX)
target_link_libraries(${CMAKE_PROJECT_NAME} ${UNIXODBC_LIBRARY} pthread)

ADD_EXECUTABLE (${TARGET} odbc_test.c)
target_link_libraries(${TARGET} ${CMAKE_DL_LIBS})
add_dependencies(${TARGET} ${CMAKE_PROJECT_NAME})
